const skyU = {
  topColor:     { value: new THREE.Color(M.skyTop) },
  bottomColor:  { value: new THREE.Color(M.skyBot) },
  horizonColor: { value: new THREE.Color(M.horizon) },
  sunDir:       { value: new THREE.Vector3(1, 1, 1).normalize() },
  sunPower:     { value: 0.0 },
  sunTint:      { value: new THREE.Color(0xfff0d0) }
};
const sky = new THREE.Mesh(
  new THREE.SphereGeometry(500, 32, 20),
  new THREE.ShaderMaterial({
    side: THREE.BackSide, depthWrite: false, fog: false, uniforms: skyU,
    vertexShader: `
      varying vec3 vDir;
      void main(){
        vec4 wp = modelMatrix * vec4(position, 1.0);
        vDir = normalize(wp.xyz - cameraPosition);
        gl_Position = projectionMatrix * viewMatrix * wp;
      }`,
    fragmentShader: `
      uniform vec3 topColor, bottomColor, horizonColor, sunTint, sunDir;
      uniform float sunPower;
      varying vec3 vDir;
      void main(){
        vec3 d = normalize(vDir);
        float h = d.y;
        vec3 low = mix(bottomColor, horizonColor, smoothstep(-0.30, 0.02, h));
        vec3 col = mix(low, topColor, smoothstep(0.0, 0.62, h));
        float sd = max(dot(d, normalize(sunDir)), 0.0);
        col += sunTint * (pow(sd, 220.0) * 1.6 + pow(sd, 8.0) * 0.28) * sunPower;
        gl_FragColor = vec4(col, 1.0);
      }`
  })
);
sky.renderOrder = -1;
scene.add(sky);

/* --------------------- 5. Procedural textures ---------------------- */

const maxAniso = renderer.capabilities.getMaxAnisotropy();
function cv(w, h){ const c = document.createElement('canvas'); c.width = w; c.height = h || w; return c; }
function tex(c, repeat){
  const t = new THREE.CanvasTexture(c);
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  if (repeat) t.repeat.set(repeat, repeat);
  t.anisotropy = Math.min(8, maxAniso);
  t.colorSpace = THREE.SRGBColorSpace;
  return t;
}

function groundTexture(){
  const c = cv(512), x = c.getContext('2d');
  x.fillStyle = '#5f5d58'; x.fillRect(0, 0, 512, 512);
  const img = x.getImageData(0, 0, 512, 512), d = img.data;
  for (let i = 0; i < d.length; i += 4){
    const n = (Math.random() - 0.5) * 38;
    d[i] += n; d[i + 1] += n; d[i + 2] += n;
  }
  x.putImageData(img, 0, 0);
  x.globalAlpha = 0.2;
  for (let i = 0; i < 240; i++){
    x.fillStyle = Math.random() > 0.5 ? '#807d76' : '#494742';
    x.beginPath(); x.arc(Math.random() * 512, Math.random() * 512, 3 + Math.random() * 20, 0, 6.283); x.fill();
  }
  x.globalAlpha = 1;
  return tex(c, 16);
}
function grassTexture(){
  const c = cv(256), x = c.getContext('2d');
  x.fillStyle = '#3f5d38'; x.fillRect(0, 0, 256, 256);
  const shades = ['#4c6f41', '#35502f', '#587c49', '#2e442a'];
  for (let i = 0; i < 3000; i++){
    x.fillStyle = shades[i & 3];
    x.fillRect(Math.random() * 256, Math.random() * 256, 2, 3 + Math.random() * 3);
  }
  return tex(c, 26);
}
function stripeTexture(a, b, n){
  const c = cv(128), x = c.getContext('2d'), w = 128 / n;
  for (let i = 0; i < n; i++){ x.fillStyle = (i % 2) ? a : b; x.fillRect(i * w, 0, w, 128); }
  return tex(c);
}
function woodTexture(base){
  const c = cv(256), x = c.getContext('2d');
  x.fillStyle = base; x.fillRect(0, 0, 256, 256);
  for (let i = 0; i < 8; i++){ x.fillStyle = 'rgba(0,0,0,.16)'; x.fillRect(0, i * 32 + 30, 256, 2); }
  for (let i = 0; i < 420; i++){
    x.strokeStyle = 'rgba(0,0,0,' + (Math.random() * 0.09).toFixed(3) + ')';
    const y = Math.random() * 256;
    x.beginPath(); x.moveTo(0, y); x.bezierCurveTo(80, y + 4, 170, y - 4, 256, y); x.stroke();
  }
  return tex(c);
}
function signTexture(text, fg, accent){
  const c = cv(512, 160), x = c.getContext('2d');
  x.fillStyle = '#0a0710'; x.fillRect(0, 0, 512, 160);
  x.strokeStyle = accent; x.lineWidth = 7; x.strokeRect(11, 11, 490, 138);
  x.textAlign = 'center'; x.textBaseline = 'middle';
  x.font = '700 62px ui-sans-serif, Helvetica, Arial, sans-serif';
  x.shadowColor = fg; x.shadowBlur = 30; x.fillStyle = fg;
  x.fillText(text, 256, 84);
  x.shadowBlur = 12; x.fillStyle = '#fffafa';
  x.fillText(text, 256, 84);
  const t = new THREE.CanvasTexture(c);
  t.colorSpace = THREE.SRGBColorSpace;
  t.anisotropy = Math.min(8, maxAniso);
  return t;
}
function glowTexture(){
  const c = cv(128), x = c.getContext('2d');
  const g = x.createRadialGradient(64, 64, 0, 64, 64, 64);
  g.addColorStop(0.00, 'rgba(255,255,255,1)');
  g.addColorStop(0.22, 'rgba(255,255,255,.72)');
  g.addColorStop(0.55, 'rgba(255,255,255,.14)');
  g.addColorStop(1.00, 'rgba(255,255,255,0)');
  x.fillStyle = g; x.fillRect(0, 0, 128, 128);
  const t = new THREE.CanvasTexture(c);
  t.colorSpace = THREE.SRGBColorSpace;
  return t;
}

// Emissive registry: intensity is driven by the active mode.
const emissives = [];
function emissiveMat(color, base, extra){
  const m = new THREE.MeshStandardMaterial(Object.assign({
    color: 0x111116, emissive: new THREE.Color(color), emissiveIntensity: base,
    roughness: 0.34, metalness: 0.1
  }, extra || {}));
  m.userData.baseEmissive = base;
  emissives.push(m);
  return m;
}
const NEON = {
  pink:   emissiveMat(0xff2e88, 1.7),
  cyan:   emissiveMat(0x24e3ff, 1.6),
  amber:  emissiveMat(0xffb020, 1.5),
  lime:   emissiveMat(0x9dff3d, 1.4),
  violet: emissiveMat(0xa45cff, 1.6),
  warm:   emissiveMat(0xffd9a0, 1.2)
};
function signMat(text, fg, accent){
  const t = signTexture(text, fg, accent);
  const m = new THREE.MeshStandardMaterial({
    map: t, emissiveMap: t, emissive: 0xffffff, emissiveIntensity: 1.1,
    roughness: 0.6, metalness: 0.0, side: THREE.DoubleSide
  });
  m.userData.baseEmissive = 1.1;
  emissives.push(m);
  return m;
}
