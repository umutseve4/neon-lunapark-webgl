// --- 7.4 Roller coaster: track, supports, station, train ---
const coaster = (() => {
  const CX = 10, CZ = 6, R = 17, SEG = 300;
  const pts = [];
  for (let i = 0; i < SEG; i++){
    const a = (i / SEG) * Math.PI * 2;
    const rr = R * (1 + 0.17 * Math.sin(3 * a));
    const y = 8.0 + 4.0 * Math.sin(2 * a) + 2.2 * Math.cos(3 * a);
    pts.push(new THREE.Vector3(CX + Math.cos(a) * rr, y, CZ + Math.sin(a) * rr * 0.92));
  }
  const curve = new THREE.CatmullRomCurve3(pts, true, 'catmullrom', 0.5);
  const UP = new THREE.Vector3(0, 1, 0);

  // Rails: two curves offset left/right of the spine.
  function offsetCurve(dist, dy){
    const out = [];
    const N = 260;
    for (let i = 0; i < N; i++){
      const u = i / N;
      const p = curve.getPointAt(u);
      const t = curve.getTangentAt(u);
      const right = new THREE.Vector3().crossVectors(t, UP).normalize();
      out.push(p.clone().addScaledVector(right, dist).setY(p.y + dy));
    }
    return new THREE.CatmullRomCurve3(out, true, 'catmullrom', 0.5);
  }
  const trackGroup = new THREE.Group();
  for (const s of [-1, 1]){
    const rail = new THREE.Mesh(new THREE.TubeGeometry(offsetCurve(s * 0.62, 0), 300, 0.11, 6, true), MAT.rail);
    rail.castShadow = true;
    trackGroup.add(rail);
  }
  // Spine tube + emissive underglow line
  const spine = new THREE.Mesh(new THREE.TubeGeometry(offsetCurve(0, -0.42), 300, 0.16, 6, true), MAT.steelDim);
  spine.castShadow = true;
  trackGroup.add(spine);
  const glowLine = new THREE.Mesh(new THREE.TubeGeometry(offsetCurve(0, -0.72), 240, 0.07, 5, true), NEON.cyan);
  trackGroup.add(glowLine);

  // Cross ties
  const TIES = 150;
  const tie = new THREE.InstancedMesh(new THREE.BoxGeometry(1.7, 0.1, 0.22), MAT.steelDim, TIES);
  tie.castShadow = true;
  const dummy = new THREE.Object3D();
  const tmp = new THREE.Vector3();
  for (let i = 0; i < TIES; i++){
    const u = i / TIES;
    const p = curve.getPointAt(u);
    const t = curve.getTangentAt(u);
    dummy.position.copy(p).setY(p.y - 0.28);
    dummy.lookAt(tmp.copy(p).add(t));
    dummy.updateMatrix();
    tie.setMatrixAt(i, dummy.matrix);
  }
  trackGroup.add(tie);

  // Supports down to the ground
  for (let i = 0; i < 34; i++){
    const u = i / 34;
    const p = curve.getPointAt(u);
    const h = p.y - 0.5;
    const col = new THREE.Mesh(new THREE.CylinderGeometry(0.16, 0.24, h, 8), MAT.steelDim);
    col.position.set(p.x, h / 2, p.z);
    col.castShadow = true;
    trackGroup.add(col);

    const foot = new THREE.Mesh(new THREE.BoxGeometry(1.1, 0.22, 1.1), MAT.white);
    foot.position.set(p.x, 0.11, p.z);
    foot.receiveShadow = true;
    trackGroup.add(foot);
  }
  park.add(trackGroup);

  // Station platform under the lowest part of the track
  let lowU = 0, lowY = Infinity;
  for (let i = 0; i < 200; i++){
    const u = i / 200, y = curve.getPointAt(u).y;
    if (y < lowY){ lowY = y; lowU = u; }
  }
  const sp = curve.getPointAt(lowU);
  const station = new THREE.Group();
  station.position.set(sp.x, 0, sp.z);
  const deck = new THREE.Mesh(new THREE.BoxGeometry(7, lowY - 1.0, 4.4), MAT.woodDark);
  deck.position.y = (lowY - 1.0) / 2;
  deck.castShadow = true; deck.receiveShadow = true;
  station.add(deck);
  const roof = new THREE.Mesh(new THREE.BoxGeometry(7.6, 0.24, 5.0), new THREE.MeshStandardMaterial({ map: stripeTexture('#e4453f', '#f5f2ec', 10), roughness: 0.7 }));
  roof.position.y = lowY + 2.4;
  roof.castShadow = true;
  station.add(roof);
  for (const sx of [-3.2, 3.2]) for (const sz of [-2.0, 2.0]){
    const p = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.11, 2.5, 8), MAT.white);
    p.position.set(sx, lowY + 1.15, sz);
    station.add(p);
  }
  const stSign = new THREE.Mesh(new THREE.PlaneGeometry(5.4, 1.7), signMat('KOMET', '#ff3b6b', '#ffd166'));
  stSign.position.set(0, lowY + 3.5, 0);
  station.add(stSign);
  park.add(station);
  addLight(0x24e3ff, 16, 26, sp.x, lowY + 2.5, sp.z, { dayScale: 0.05, hue: 0.52 });

  // Train
  const train = new THREE.Group();
  const cars = [];
  const CARS = 5;
  const bodyMat = new THREE.MeshStandardMaterial({ color: 0xe23b48, roughness: 0.4, metalness: 0.35 });
  const headMat = new THREE.MeshStandardMaterial({ color: 0xffd166, roughness: 0.4, metalness: 0.3 });
  const riderColors = [0x2f5fa8, 0x1fa97a, 0xe0913a, 0xb44bd6, 0xd94f6b];
  for (let i = 0; i < CARS; i++){
    const car = new THREE.Group();

    const body = new THREE.Mesh(new THREE.BoxGeometry(1.5, 0.78, 2.1), i === 0 ? headMat : bodyMat);
    body.position.y = 0.62;
    body.castShadow = true;
    car.add(body);

    const nose = new THREE.Mesh(new THREE.ConeGeometry(0.62, 1.1, 12), i === 0 ? headMat : bodyMat);
    nose.rotation.x = Math.PI / 2;
    nose.position.set(0, 0.62, 1.5);
    nose.castShadow = true;
    car.add(nose);

    const strip = new THREE.Mesh(new THREE.BoxGeometry(1.56, 0.12, 2.14), i === 0 ? NEON.amber : NEON.cyan);
    strip.position.y = 0.34;
    car.add(strip);

    const chassis = new THREE.Mesh(new THREE.BoxGeometry(1.0, 0.2, 1.6), MAT.steelDim);
    chassis.position.y = 0.12;
    car.add(chassis);

    for (const sx of [-0.8, 0.8]) for (const sz of [-0.72, 0.72]){
      const w = new THREE.Mesh(new THREE.CylinderGeometry(0.2, 0.2, 0.14, 10), MAT.steel);
      w.rotation.z = Math.PI / 2;
      w.position.set(sx, 0.16, sz);
      car.add(w);
    }

    // Two riders per car
    for (const sx of [-0.34, 0.34]){
      const head = new THREE.Mesh(new THREE.SphereGeometry(0.2, 10, 8),
        new THREE.MeshStandardMaterial({ color: riderColors[(i + (sx > 0 ? 1 : 0)) % riderColors.length], roughness: 0.8 }));
      head.position.set(sx, 1.18, -0.15);
      head.castShadow = true;
      car.add(head);
      const arm = new THREE.Mesh(new THREE.CylinderGeometry(0.06, 0.06, 0.8, 6),
        new THREE.MeshStandardMaterial({ color: 0xf0c9a0, roughness: 0.9 }));
      arm.position.set(sx, 1.35, -0.1);
      arm.rotation.z = sx > 0 ? -0.25 : 0.25;
      car.add(arm);
    }

    train.add(car);
    cars.push(car);
  }
  park.add(train);

  return { curve, cars, spacing: 0.0135, length: curve.getLength() };
})();
