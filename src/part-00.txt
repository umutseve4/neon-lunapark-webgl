/* Quality tier — phones and coarse-pointer devices pay for DPR², bloom and
   shadow maps at the same time, so cap all three up front. */
const COARSE  = window.matchMedia('(pointer: coarse)').matches;
const SMALL   = Math.min(window.screen ? window.screen.width : 1920,
                         window.screen ? window.screen.height : 1080) < 820;
const LOWPWR  = COARSE || SMALL;
const SHADOW_RES  = LOWPWR ? 1024 : 2048;
const BLOOM_SCALE = LOWPWR ? 0.5 : 1;
let   dprCap      = LOWPWR ? 1.25 : 2;
const dpr = () => Math.min(window.devicePixelRatio || 1, dprCap);

function say(msg){
  // Screen-reader announcement; re-set to '' first so repeats are still spoken.
  if (!sayEl) return;
  sayEl.textContent = '';
  setTimeout(() => { sayEl.textContent = msg; }, 30);
}

function fail(err){
  console.error(err);
  boot.classList.remove('gone');
  const bar = boot.querySelector('.bar'); if (bar) bar.remove();
  bootMsg.innerHTML = '<div class="err"><b>Sahne başlatılamadı.</b><br>WebGL desteği ya da CDN erişimi yok gibi görünüyor.<br>' +
    String(err && err.message ? err.message : err).replace(/[<>&]/g, ' ') + '</div>';
  if (!boot.querySelector('.reload')){
    const b = document.createElement('button');
    b.type = 'button'; b.className = 'reload'; b.textContent = 'Yeniden yükle';
    b.addEventListener('click', () => location.reload());
    bootMsg.appendChild(b);
    b.focus();
  }
}
window.addEventListener('error', e => { if (!boot.classList.contains('gone')) fail(e.error || e.message); });
