function buildBooth(cfg){

  g.position.set(cfg.x, 0, cfg.z);
  g.rotation.y = cfg.ry || 0;

  const W = cfg.w || 6, D = cfg.d || 4, H = cfg.h || 3.2;

  const body = new THREE.Mesh(new THREE.BoxGeometry(W, H, D), cfg.bodyMat || MAT.wood);
  body.position.y = H / 2;
  body.castShadow = true; body.receiveShadow = true;
  g.add(body);

  const counter = new THREE.Mesh(new THREE.BoxGeometry(W + 0.7, 0.22, 1.0), MAT.white);
  counter.position.set(0, H * 0.55, D / 2 + 0.4);
  counter.castShadow = true;
  g.add(counter);

  win.position.set(0, H * 0.68, D / 2 + 0.011);
  g.add(win);

  // Striped awning
  const awn = new THREE.Mesh(
    new THREE.BoxGeometry(W + 1.2, 0.16, 2.4),
    new THREE.MeshStandardMaterial({ map: stripeTexture(cfg.stripeA, cfg.stripeB, 12), roughness: 0.7 })
  );
  awn.position.set(0, H + 0.35, D / 2 + 0.9);
  awn.rotation.x = -0.22;
  awn.castShadow = true;
  g.add(awn);

  for (const sx of [-(W / 2 + 0.4), (W / 2 + 0.4)]){
    const p = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.09, H + 0.2, 8), MAT.steel);
    p.position.set(sx, (H + 0.2) / 2, D / 2 + 1.9);
    g.add(p);
  }

  // Neon sign board on the roof
  const boardH = 1.7, boardW = Math.min(W + 0.8, 6.6);
  const sign = new THREE.Mesh(new THREE.PlaneGeometry(boardW, boardH), signMat(cfg.sign, cfg.signFg, cfg.signAccent));
  sign.position.set(0, H + 1.5, D / 2 - 0.2);
  g.add(sign);

  const frame = new THREE.Mesh(new THREE.TorusGeometry(boardW * 0.52, 0.07, 6, 4, Math.PI * 2), cfg.neon);
  frame.rotation.z = Math.PI / 4;

  const trim = new THREE.Mesh(new THREE.BoxGeometry(W + 0.3, 0.12, D + 0.3), cfg.neon);
  trim.position.y = H + 0.05;
  g.add(trim);

  park.add(g);
  return g;
}

// --- 7.8 Lamp posts + hanging string lights ---
const bulbs = (() => {
  const positions = [];
  const colors = [];
  const phases = [];
  const palette = [
    new THREE.Color(0xff2e88), new THREE.Color(0x24e3ff), new THREE.Color(0xffd166),
    new THREE.Color(0x9dff3d), new THREE.Color(0xa45cff), new THREE.Color(0xff8b3d)
  ];

  const POSTS = 14, PR = 43, PH = 7.0;
  const anchors = [];
  for (let i = 0; i < POSTS; i++){
    const a = (i / POSTS) * Math.PI * 2 + 0.12;
    const x = Math.cos(a) * PR, z = Math.sin(a) * PR;
    anchors.push(new THREE.Vector3(x, PH, z));

    const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.13, 0.2, PH, 8), MAT.steelDim);
    pole.position.set(x, PH / 2, z);
    pole.castShadow = true;
    park.add(pole);

    const head = new THREE.Mesh(new THREE.SphereGeometry(0.42, 12, 10), NEON.warm);
    head.position.set(x, PH + 0.25, z);
    park.add(head);

    const cap = new THREE.Mesh(new THREE.ConeGeometry(0.75, 0.5, 10), MAT.steelDim);
    cap.position.set(x, PH + 0.72, z);
    park.add(cap);
  }

  // Catenary strings between neighbouring posts
  for (let i = 0; i < POSTS; i++){
    const a = anchors[i], b = anchors[(i + 1) % POSTS];
    const N = 13;
    for (let k = 1; k < N; k++){
      const t = k / N;
      const p = new THREE.Vector3().lerpVectors(a, b, t);
      p.y -= Math.sin(t * Math.PI) * 2.0;
      positions.push(p.x, p.y, p.z);
      const c = palette[(i + k) % palette.length];
      colors.push(c.r, c.g, c.b);
      phases.push(Math.random() * Math.PI * 2);
    }
  }

  // Extra strings radiating from the entrance arch to the plaza
  const hub = new THREE.Vector3(0, 11.5, 45);
  for (let s = 0; s < 4; s++){
    const end = new THREE.Vector3(-14 + s * 9.5, 6.5, 26);
    for (let k = 1; k < 12; k++){
      const t = k / 12;
      const p = new THREE.Vector3().lerpVectors(hub, end, t);
      p.y -= Math.sin(t * Math.PI) * 1.6;
      positions.push(p.x, p.y, p.z);
      const c = palette[(s + k) % palette.length];
      colors.push(c.r, c.g, c.b);
      phases.push(Math.random() * Math.PI * 2);
    }
  }

  const geo = new THREE.BufferGeometry();
  const posAttr = new THREE.Float32BufferAttribute(positions, 3);
  const colAttr = new THREE.Float32BufferAttribute(colors, 3);
  geo.setAttribute('position', posAttr);
  geo.setAttribute('color', colAttr);
  const mat = new THREE.PointsMaterial({
    size: 1.15, map: GLOW, vertexColors: true, transparent: true,
    depthWrite: false, blending: THREE.AdditiveBlending, sizeAttenuation: true, opacity: 1
  });
  const pts = new THREE.Points(geo, mat);
  pts.frustumCulled = false;
  park.add(pts);

  return { points: pts, material: mat, base: new Float32Array(colors), colAttr, phases, count: phases.length };
})();
